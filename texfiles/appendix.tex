% The \textsc{Guards-Empty} rule leaves the policy unchanged when no guardrail is triggered. The \textsc{Guards-Step} rule applies one triggered guardrail action and continues with the rest, thereby modeling simultaneous execution as an arbitrary serialization of corrective actions. The \textsc{Guarded-Transition} rule then composes this update phase with the underlying system step: the system first moves from $\sigma$ to $\sigma'$ under policy $\pi$, after which the triggered guardrails jointly determine the policy $\pi'$ used on subsequent steps.

% \paragraph{Selected-guard semantics.}
Once a guardrail has been selected into the current repair phase, its execution
does not re-check enabledness on the accumulator policy. We therefore use the
following auxiliary relation:
\[
G \models^{\mathsf{act}}_{\sigma} \hat\pi \to \hat\pi'
\]
which means that guardrail $G$, having already been selected into the active set,
executes on post-state $\sigma$ and accumulator policy $\hat\pi$, producing
$\hat\pi'$. Formally:
\begin{mathpar}

}{

}

    \neg P(\sigma)

\paragraph{Frozen-active-set guarded semantics.}
Let
\[
T(\sigma',\pi) \;\defeq\; \{\, G_i \in \mathcal G \mid E_i(\pi)=1 \wedge \neg P_i(\sigma') \,\}
\]
be the triggered set computed from the original policy $\pi$ and the shared
post-state $\sigma'$. We write
\[
\mathcal G \models^{\mathsf{act}}_{\sigma',\pi} \hat\pi \to \hat\pi'
\]
to mean that the guardrails selected in $T(\sigma',\pi)$ transform the current
accumulator policy $\hat\pi$ into $\hat\pi'$ under an arbitrary serialization.
This relation is defined inductively by:
\begin{mathpar}
\inferrule*[right=Actives-Empty]
  { }
  { \emptyset \models^{\mathsf{act}}_{\sigma',\pi} \hat\pi \to \hat\pi }

\inferrule*[right=Actives-Step]
  { G \in T(\sigma',\pi) \\
    G \models^{\mathsf{act}}_{\sigma'} \hat\pi \to \hat\pi_1 \\
    \mathcal G \setminus \{G\} \models^{\mathsf{act}}_{\sigma',\pi} \hat\pi_1 \to \hat\pi' }

\section{Verification Soundness}
\label{appdx:verification-soundness}
For each guardrail $G_i = (\property{i},\action{i},\enabled_i)$ and chosen local
guarantee $\varphi_i$, we assume:
\[
\begin{array}{@{}c@{\qquad}l@{}}
M(s,\pi,s') \wedge E_i(\pi) \wedge P_i(s')
\Rightarrow
\varphi_i(s',\pi)
&
\textsc{Maintain}_i
\\[0.8em]
\begin{array}{@{}c@{}}
M(s,\pi,s') \wedge E_i(\pi) \wedge \neg P_i(s')
\\
\wedge\; G_i \models^{\mathsf{act}}_{s'} \hat\pi \to \hat\pi'
\end{array}
\Rightarrow
\varphi_i(s',\hat\pi')
&
\textsc{Recover}_i
\end{array}
\]

\paragraph{Compatibility obligation.}
For each distinct pair of guardrails $G_i$ and $G_j$, we assume:
\[
\begin{array}{@{}c@{}}
E_i(\pi)\wedge \neg P_i(s)\wedge \varphi_j(s,\hat\pi)
\\
\wedge G_i \models^{\mathsf{act}}_s \hat\pi \to \hat\pi'
\end{array}
\Rightarrow
\varphi_j(s,\hat\pi')
\qquad
\textsc{Compat}_{i,j}
\]
This says that once the active set has been fixed, executing one selected
guardrail preserves the guarantees already established for the others.

\paragraph{Active-set invariant.}
Since enabledness is determined from the input policy that selected the active
set, the relevant conjunction of local guarantees must be parameterized by that
policy. For any input policy $\pi_0$, define
\begin{equation}
I_{\pi_0}(s,\hat\pi)\;\defeq\;\bigwedge_{i=1}^n \bigl(E_i(\pi_0)\Rightarrow \varphi_i(s,\hat\pi)\bigr).
\label{eq:active-set-invariant}
\end{equation}
The corresponding global proof obligation is
\[
\forall \pi_0,s,\hat\pi.\;
I_{\pi_0}(s,\hat\pi)\Rightarrow \Phi(s,\hat\pi).
\]

\begin{theorem}[One-Step Soundness Under Frozen Active Sets]
\label{thm:frozen-active-set-soundness}
Assume every installed guardrail satisfies \textsc{Maintain} and
\textsc{Recover}, every distinct pair satisfies \textsc{Compat}, and
\[
\forall \pi_0,s,\hat\pi.\;
I_{\pi_0}(s,\hat\pi)\Rightarrow \Phi(s,\hat\pi).
\]
If
\[
\mathcal G \models^{\mathsf{act}} \langle s,\pi\rangle \rightsquigarrow \langle s',\pi'\rangle,
\]
then
\[
\Phi(s',\pi').
\]
\end{theorem}

\begin{proof}
Consider an arbitrary guarded transition
\[
\mathcal G \models^{\mathsf{act}} \langle s,\pi\rangle \rightsquigarrow \langle s',\pi'\rangle.
\]
By \textsc{Guarded-Trans}$_{\mathsf{act}}$, there is a base system step
\[
s \xrightarrow{\pi} s'
\]
followed by a serialization of the triggered guardrails computed from the pair
$(s',\pi)$. Let
\[
T(s',\pi)=\{G_{i_1},\ldots,G_{i_m}\}
\]
be the triggered set, ordered according to the serialization chosen in this
transition, and let
\[
\pi_0,\pi_1,\ldots,\pi_m
\]
be the corresponding sequence of accumulator policies, where $\pi_0=\pi$,
$\pi_m=\pi'$, and for each $\ell \in \{1,\ldots,m\}$,
\[
G_{i_\ell}\models^{\mathsf{act}}_{s'} \pi_{\ell-1}\to \pi_\ell .
\]

Let
\[
U \;\defeq\; \{\, j \mid E_j(\pi)=1 \text{ and } P_j(s') \,\}.
\]
Thus, $U$ is exactly the set of guardrails enabled on the input policy whose
monitored conditions already hold on the shared post-state.

For each $j \in U$, \textsc{Maintain}$_j$ applied to the base step
$M(s,\pi,s')$ yields
\[
\varphi_j(s',\pi_0).
\]
We now show, by induction on $\ell \in [0,m]$, that after processing the first
$\ell$ selected guardrails, every guardrail in
\[
U \cup \{i_1,\ldots,i_\ell\}
\]
satisfies its local guarantee at $(s',\pi_\ell)$.

\medskip
\noindent
\textbf{Base case $(\ell=0)$.}
This is exactly the statement established above for all $j \in U$.

\medskip
\noindent
\textbf{Inductive step.}
Assume the claim holds for $\ell-1$, where $1 \le \ell \le m$. We show it for
$\ell$.
Since $G_{i_\ell}\in T(s',\pi)$, we have $E_{i_\ell}(\pi)=1$ and
$\neg P_{i_\ell}(s')$. By construction of the serialization,
\[
G_{i_\ell}\models^{\mathsf{act}}_{s'} \pi_{\ell-1}\to \pi_\ell .
\]
Applying \textsc{Recover}$_{i_\ell}$ to this selected-guard step yields
\[
\varphi_{i_\ell}(s',\pi_\ell).
\]

Now let $j \in U \cup \{i_1,\ldots,i_{\ell-1}\}$. By the induction hypothesis,
\[
\varphi_j(s',\pi_{\ell-1})
\]
already holds. Using the same selected-guard step
\[
G_{i_\ell}\models^{\mathsf{act}}_{s'} \pi_{\ell-1}\to \pi_\ell
\]
and \textsc{Compat}$_{i_\ell,j}$, we conclude that executing $G_{i_\ell}$
preserves $\varphi_j$, so
\[
\varphi_j(s',\pi_\ell)
\]
also holds.

Thus, after processing $G_{i_\ell}$, every guardrail in
\[
U \cup \{i_1,\ldots,i_\ell\}
\]
satisfies its local guarantee at $(s',\pi_\ell)$, completing the inductive
step.

Taking $\ell=m$, we conclude that every guardrail in
\[
U \cup \{i_1,\ldots,i_m\}
\]
satisfies its local guarantee at $(s',\pi_m)$.
By definition of $U$ and $T(s',\pi)$,
\[
U \cup \{i_1,\ldots,i_m\}
\;=\;
\{\, j \mid E_j(\pi)=1 \,\},
\]
because every guardrail enabled on the input policy either satisfies its
property on $s'$ and belongs to $U$, or violates its property on $s'$ and
belongs to $T(s',\pi)$.
Since $\pi_m=\pi'$, it follows that
\[
I_\pi(s',\pi')
\]
holds.
The global proof obligation therefore yields
\[
\Phi(s',\pi'),
\]
as required.
\end{proof}

\begin{corollary}[Reachable-State Soundness]
\label{cor:frozen-active-set-reachability}
Assume the hypotheses of
Theorem~\ref{thm:frozen-active-set-soundness}, and assume the initial runtime
state $\langle s_0,\pi_0\rangle$ satisfies
\[
I_{\pi_0}(s_0,\pi_0).
\]
Then every state reachable under
\[
\mathcal G \models^{\mathsf{act}} \langle s,\pi\rangle \rightsquigarrow \langle s',\pi'\rangle
\]
satisfies $\Phi$.
\end{corollary}

\begin{proof}
For the initial state, $I_{\pi_0}(s_0,\pi_0)$ and the global proof obligation
immediately imply $\Phi(s_0,\pi_0)$.
Now consider any reachable state obtained after at least one guarded transition.
Such a state is the target of some transition
\[
\mathcal G \models^{\mathsf{act}} \langle s,\pi\rangle \rightsquigarrow \langle s',\pi'\rangle.
\]
By Theorem~\ref{thm:frozen-active-set-soundness}, this implies
\[
\Phi(s',\pi').
\]
Hence every reachable state satisfies $\Phi$.
\end{proof}
